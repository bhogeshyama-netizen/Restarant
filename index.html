Apps Script Code
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data;

  try {
    data = JSON.parse(e.postData.contents);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'error', message: 'Invalid JSON' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Skip test pings — don't write them to sheet
  if (data.orderId === 'TEST-PING') {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'success', message: 'Connection OK' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Add header if sheet is empty
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      'Order ID', 'Date', 'Time',
      'Item', 'Type', 'Qty',
      'Unit Price', 'Line Total', 'Order Total'
    ]);
    sheet.getRange(1, 1, 1, 9).setFontWeight('bold');
  }

  var orderId = data.orderId;
  var date = data.date;
  var time = data.time;
  var orderTotal = data.total;

  data.items.forEach(function(item) {
    sheet.appendRow([
      orderId, date, time,
      item.name, item.type, item.qty,
      item.price, item.qty * item.price,
      orderTotal
    ]);
  });

  return ContentService
    .createTextOutput(JSON.stringify({
      status: 'success',
      orderId: orderId
    }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  return ContentService
    .createTextOutput(JSON.stringify({
      status: 'success',
      message: 'Connection OK'
    }))
    .setMimeType(ContentService.MimeType.JSON);
}
